\begin{frame}{Vì sao phải nhìn SCRFD theo pipeline}
  \begin{itemize}
    \item Nếu chỉ nhìn \textbf{tổng FLOPs}, ta không trả lời được vì sao cùng khoảng 2.5 GFLOPs nhưng Hard AP chênh nhau vài điểm.
    \item Nếu chỉ nhìn \textbf{augmentation}, ta cũng không giải thích được vì sao cùng scale set nhưng backbone này học tốt hơn backbone kia.
    \item SCRFD hiệu quả vì nó tối ưu đồng thời:
  \end{itemize}
  \vspace{0.2cm}
  \begin{block}{Ba đối tượng tối ưu}
    \begin{enumerate}
      \item \textbf{Kiến trúc}: compute rơi vào stage nào.
      \item \textbf{Dữ liệu huấn luyện}: tiny faces xuất hiện với tần suất nào sau crop/resize.
      \item \textbf{Supervision}: tiny faces có nhận đủ positive anchors hay không.
    \end{enumerate}
  \end{block}
\end{frame}

\section{SCRFD Gốc}

\begin{frame}{Kiến trúc nền của SCRFD}
  \begin{columns}[T,onlytextwidth]
    \column{0.57\textwidth}
    \includegraphics[width=\linewidth]{scrfd_backbone_neck_head.png}
    \column{0.43\textwidth}
    \begin{itemize}
      \item \textbf{Backbone}: tạo chuỗi đặc trưng \(C_2 \rightarrow C_5\).
      \item \textbf{Neck}: hợp nhất thông tin đa tỉ lệ để giữ cả ngữ nghĩa lẫn vị trí.
      \item \textbf{Head}: phân loại khuôn mặt/nền và hồi quy bbox trên các feature map.
      \item Điểm mấu chốt: face detection, đặc biệt là \textbf{tiny faces}, phụ thuộc mạnh vào các tầng nông có độ phân giải cao.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Computation Redistribution: flow làm việc thực sự}
  \begin{enumerate}
    \item Chọn một detector template trong miền 2.5 GFLOPs.
    \item Lấy mẫu \textbf{độ sâu, độ rộng và tỉ lệ kênh} của các stage backbone; sau đó mở rộng sang neck và head.
    \item Tính lại FLOPs cho candidate; chỉ giữ những cấu hình còn nằm trong \textbf{cùng dải chi phí} quanh mục tiêu 2.5 GFLOPs.
    \item Train từng candidate đủ để so sánh, rồi đo AP trên WIDER FACE.
    \item Chọn winner theo chất lượng thực nghiệm, không theo trực giác thủ công.
  \end{enumerate}
  \vspace{0.25cm}
  \begin{block}{Ý nghĩa}
    CR biến việc “thiết kế detector” thành một bài toán \textbf{sinh -- lọc -- huấn luyện -- đánh giá} dưới cùng ràng buộc chi phí.
  \end{block}
\end{frame}

\begin{frame}{CR rút ra điều gì cho face detection}
  \begin{itemize}
    \item Detector tốt không đơn giản là detector “to hơn”, mà là detector \textbf{đặt FLOPs đúng chỗ hơn}.
    \item Với tiny faces, stage nông vừa quý vừa đắt: tăng compute ở đây giúp giữ chi tiết không gian, nhưng nếu tăng sai chỗ sẽ làm chi phí bùng lên.
    \item Vì vậy CR không đi theo quy tắc cứng kiểu “luôn tăng sớm” hay “luôn tăng sâu”, mà tìm \textbf{tỉ lệ phân bổ hợp lý} giữa \(C_2,C_3,C_4,C_5\), neck và head.
    \item Đây cũng chính là điểm mà phần \textbf{SPOS-NAS} của nhóm sẽ đi sâu phân tích lại.
  \end{itemize}
\end{frame}

\begin{frame}{Sample Redistribution: flow trong pipeline huấn luyện}
  \begin{enumerate}
    \item Mỗi ảnh train được chọn một \textbf{scale crop} từ scale set.
    \item Ảnh bị crop thành patch vuông rồi resize về kích thước cố định \(640 \times 640\).
    \item Cùng một khuôn mặt, sau bước này có thể trở nên \textbf{lớn hơn} hoặc \textbf{nhỏ hơn} tương đối so với anchor và feature map.
    \item Bước gán mẫu dương/âm vì thế thay đổi: tiny faces có thể nhận nhiều positive hơn hoặc ít positive hơn.
    \item Chính điều đó làm SR ảnh hưởng trực tiếp tới Hard AP.
  \end{enumerate}
  \vspace{0.2cm}
  \begin{block}{Luồng nhân quả}
    scale set \(\rightarrow\) kích thước mặt sau resize \(\rightarrow\) positive anchors \(\rightarrow\) training signal \(\rightarrow\) Hard AP.
  \end{block}
\end{frame}

\begin{frame}{SR không chỉ là augmentation thông thường}
  \begin{itemize}
    \item Nếu scale set thiên về zoom-in, detector thấy nhiều mặt vừa/lớn hơn nhưng ít tình huống tiny faces thật sự.
    \item Nếu scale set tạo thêm zoom-out hợp lý, tiny faces xuất hiện dày hơn sau resize và detector học đúng vùng khó của WIDER FACE.
    \item Trong reproduction của nhóm, chỉ riêng việc đổi từ default scale set sang \textbf{paper-SR12} đã làm Hard AP tăng từ \textbf{72.49} lên \textbf{73.71}.
    \item Nghĩa là cùng một kiến trúc SCRFD-2.5GF, \textbf{phân bố sample khi train} đã đủ tạo chênh lệch rõ rệt.
  \end{itemize}
\end{frame}

\section{SPOS-NAS Cho CR}

\begin{frame}{Vì sao dùng SPOS-NAS để phân tích CR}
  \begin{itemize}
    \item Official SCRFD dùng empirical search tốn tài nguyên; nhóm cần một cách \textbf{nhẹ hơn} để kiểm tra lại trực giác phân bổ compute.
    \item SPOS-NAS cho phép huấn luyện \textbf{một supernet duy nhất}, sau đó rút ra nhiều candidate bằng inherited weights.
    \item Mục tiêu của phần này không thay thế official code, mà trả lời câu hỏi:
  \end{itemize}
  \begin{block}{Câu hỏi}
    Trong cùng khoảng 2.5 GFLOPs, winner có thật sự dồn compute vào những stage hữu ích cho face detection hay không?
  \end{block}
\end{frame}

\begin{frame}{Flow triển khai SPOS-NAS proxy}
  \begin{enumerate}
    \item Xây dựng supernet bao phủ nhiều lựa chọn depth/width trên bốn stage \(C_2 \rightarrow C_5\).
    \item Mỗi mini-batch chỉ kích hoạt \textbf{một đường con} để huấn luyện theo tinh thần single-path.
    \item Sau khi supernet học đủ ổn định, lấy mẫu nhiều candidate trong dải FLOPs mục tiêu.
    \item Với mỗi candidate, chạy \textbf{calibration thống kê chuẩn hóa}, rồi đo proxy score trên tập validation.
    \item Xếp hạng 1\,000 kiến trúc hợp lệ và phân tích cấu trúc của nhóm đứng đầu.
  \end{enumerate}
\end{frame}

\begin{frame}{Kết quả chính của SPOS-NAS proxy}
  \small
  \begin{tabular}{@{}p{0.24\textwidth}p{0.66\textwidth}@{}}
    \toprule
    Quan sát & Ý nghĩa \\
    \midrule
    Supernet đạt best validation loss khoảng \textbf{0.6519} ở epoch 18 & Proxy đủ ổn định để so sánh candidate trong giới hạn tài nguyên \\
    Đánh giá \textbf{1\,000} kiến trúc trong cùng vùng 2.5 GFLOPs & So sánh được pattern phân bổ compute thay vì chỉ một vài mẫu ngẫu nhiên \\
    Top group không đi theo phân bổ đồng đều & Winner có cấu trúc rõ ràng, không phải may mắn thống kê \\
    \bottomrule
  \end{tabular}

  \vspace{0.35cm}
  \begin{itemize}
    \item Điều nhóm cần không chỉ là top-1, mà là \textbf{mẫu hình chung} của Top-100 so với nhóm thua.
  \end{itemize}
\end{frame}

\begin{frame}{Insight lớn nhất từ SPOS-NAS}
  \begin{itemize}
    \item Top-100 winners có xu hướng \textbf{đầu tư ổn định vào \(C_2\) và \(C_5\)} thay vì dồn nhiều compute vào \(C_3\).
    \item \(C_2\) giữ chi tiết không gian cho tiny faces; \(C_5\) bổ sung ngữ nghĩa đủ mạnh cho head.
    \item \(C_3\) thường là \textbf{bẫy chi phí}: feature map vẫn còn lớn nên FLOPs đắt, nhưng lợi ích tăng không tương xứng nếu phình quá mức.
    \item Khi proxy target đổi từ \textbf{center-focus} sang \textbf{edge-focus}, compute dịch thêm khỏi \(C_3\) và trải về \(C_2/C_5\), càng củng cố kết luận trên.
  \end{itemize}
\end{frame}

\begin{frame}{Kết luận của phần SPOS-NAS}
  \begin{block}{Điều SPOS-NAS xác nhận}
    Computation Redistribution của SCRFD là một ý tưởng \textbf{có cấu trúc thật}, không phải một mẹo search ngẫu nhiên.
  \end{block}
  \begin{itemize}
    \item Trong miền 2.5 GFLOPs, winner không thắng vì “to hơn”, mà vì \textbf{phân bổ compute đúng hơn}.
    \item Quy luật rút ra là: giữ đủ năng lực ở tầng nông để không mất tiny faces, đồng thời để tầng sâu đủ mạnh cho ngữ nghĩa; tránh phình quá mức ở vùng trung gian đắt đỏ.
  \end{itemize}
\end{frame}

\section{ASR+JSAR Cho SR Và Train}

\begin{frame}{Vì sao cần ASR+JSAR sau baseline}
  \begin{itemize}
    \item Baseline paper-SR12 đã tốt hơn default scale set, nhưng Hard AP vẫn còn cách official model zoo hơn \textbf{4 điểm}.
    \item Điều này gợi ý rằng scale set tĩnh vẫn chưa đủ: detector chưa nhìn thấy tiny faces với \textbf{độ khó đúng mức}, và bước assignment vẫn có thể bỏ sót positive samples.
    \item Vì vậy nhóm cải tiến theo hai hướng nối tiếp:
  \end{itemize}
  \begin{block}{Hai đòn can thiệp}
    \textbf{ASR}: sửa cách chọn scale khi train. \\
    \textbf{JSAR}: sửa cách tiny faces nhận positive anchors.
  \end{block}
\end{frame}

\begin{frame}{ASR: flow cập nhật phân bố scale theo độ khó}
  \begin{enumerate}
    \item Bắt đầu từ một scale pool đủ rộng, không khóa cứng xác suất như paper-SR12.
    \item Sau mỗi giai đoạn huấn luyện, gom thống kê kích thước mặt và mức độ thiếu hụt recall trên các vùng khó.
    \item Chia các mặt vào các bin kích thước; bin nào đang thiếu supervision sẽ được tăng xác suất lấy mẫu.
    \item Làm mượt thống kê theo kiểu \textbf{EMA} để policy không giật giữa các epoch.
    \item Epoch sau dùng phân bố mới để tạo crop/resize, từ đó đưa nhiều tiny faces có ích hơn vào train.
  \end{enumerate}
\end{frame}

\begin{frame}{JSAR: flow tăng positive assignment cho tiny faces}
  \begin{enumerate}
    \item Với các ground truth rất nhỏ, điều kiện gán dương mặc định thường quá chặt nên số positive bị thiếu.
    \item JSAR nới lỏng việc chọn candidate quanh tâm và giảm mức khắt khe của bước gán cho nhóm tiny faces.
    \item Nếu sau các bước chuẩn mà một tiny face vẫn gần như không có positive, cơ chế \textbf{fallback} sẽ cưỡng bức giữ lại một nhóm anchor gần nhất còn hợp lý.
    \item Nhờ vậy, tiny faces không còn bị “rớt supervision” ở cuối pipeline train.
  \end{enumerate}
  \vspace{0.2cm}
  \begin{block}{Vai trò riêng}
    ASR sửa \textbf{đầu vào huấn luyện}; JSAR sửa \textbf{bước nhận tín hiệu học}.
  \end{block}
\end{frame}

\begin{frame}{Kết quả ASR+JSAR trên WIDER FACE}
  \small
  \begin{tabular}{@{}lcccc@{}}
    \toprule
    Thiết lập & Easy & Medium & Hard & mAP \\
    \midrule
    Baseline default scale set & 93.62 & 91.81 & 72.49 & 85.97 \\
    Baseline paper-SR12 & 93.82 & 91.90 & 73.71 & 86.48 \\
    ASR+JSAR + paper-SR12 & 94.10 & 92.58 & 77.38 & 88.02 \\
    Official model zoo & 93.78 & 92.16 & 77.87 & 87.94 \\
    \bottomrule
  \end{tabular}

  \vspace{0.35cm}
  \begin{itemize}
    \item Mức tăng quan trọng nhất nằm ở \textbf{Hard}: từ \textbf{73.71} lên \textbf{77.38}.
    \item Tức là cải tiến của nhóm đã kéo reproduction tới rất gần mức official ở split khó nhất.
  \end{itemize}
\end{frame}

\begin{frame}{Vì sao ASR+JSAR có tác dụng}
  \begin{itemize}
    \item ASR làm tiny faces xuất hiện với \textbf{phân bố phù hợp hơn} trong các epoch cuối, nên head được nhìn thấy nhiều tình huống khó hơn.
    \item JSAR làm số positive trên mỗi tiny ground truth tăng lên rõ rệt; theo thống kê của nhóm, \textbf{positives per GT} tăng từ khoảng \textbf{1.76} lên \textbf{2.67}.
    \item Hai cơ chế này bổ sung nhau:
  \end{itemize}
  \begin{block}{Phối hợp}
    ASR đưa đúng dữ liệu khó vào train; JSAR đảm bảo dữ liệu khó đó \textbf{thực sự sinh ra supervision}.
  \end{block}
\end{frame}

\begin{frame}{Kết luận của phần ASR+JSAR}
  \begin{itemize}
    \item Nếu chỉ thay scale set, detector tiến bộ nhưng chưa đủ.
    \item Nếu chỉ sửa assignment mà dữ liệu train vẫn thiếu tiny faces, hiệu quả cũng bị giới hạn.
    \item Hard AP tăng mạnh chỉ khi \textbf{phân bố sample} và \textbf{cơ chế nhận mẫu dương} được sửa cùng lúc.
    \item Đây là phần nối trực tiếp giữa \textbf{Sample Redistribution} của paper và hành vi huấn luyện quan sát được trong code.
  \end{itemize}
\end{frame}

\section{Kết Luận}

\begin{frame}{Kết luận}
  \begin{itemize}
    \item \textbf{SCRFD mạnh không chỉ vì tổng FLOPs}, mà vì FLOPs được phân bổ đúng chỗ trong detector.
    \item \textbf{SPOS-NAS} xác nhận trực giác của CR: winner trong miền 2.5 GFLOPs có phân bổ compute có cấu trúc, đặc biệt tránh đầu tư quá mức vào \(C_3\).
    \item \textbf{ASR+JSAR} cho thấy phần khó còn lại nằm ở training signal cho tiny faces: sửa scale distribution và assignment có thể kéo Hard AP gần tới official result.
    \item Insight tổng quát của báo cáo là: với face detection hiệu quả, phải tối ưu đồng thời \textbf{kiến trúc}, \textbf{dữ liệu huấn luyện} và \textbf{supervision}.
  \end{itemize}
\end{frame}
